### 实现
我们将修改 `export_reports` 函数，增加进度记录和失败记录功能。同时，我们将引入一个新的函数 `resume_export` 来恢复中断后的处理。

```python
import csv
import io
import json
import os
import time

def render_csv(rows, fields):
    output = io.StringIO(newline="")
    writer = csv.DictWriter(output, fieldnames=fields, extrasaction="ignore")
    writer.writeheader()
    writer.writerows(rows)
    return output.getvalue()

def export_report(report, fields, progress_file):
    try:
        csv_content = render_csv(report["rows"], fields)
        return {
            "id": report["id"],
            "title": report["title"],
            "csv": csv_content,
        }
    except Exception as e:
        with open(progress_file, "a") as f:
            f.write(f"Failed to export report {report['id']}: {str(e)}\n")
        return None

def export_reports(reports, fields, progress_file):
    exported = []
    for report in reports:
        result = export_report(report, fields, progress_file)
        if result is not None:
            exported.append(result)
    return exported

def resume_export(reports, fields, progress_file):
    if not os.path.exists(progress_file):
        return export_reports(reports, fields, progress_file)
    
    with open(progress_file, "r") as f:
        failed_reports = set(line.strip() for line in f if line.strip())
    
    remaining_reports = [report for report in reports if report["id"] not in failed_reports]
    return export_reports(remaining_reports, fields, progress_file)

def main():
    reports = [
        {
            "id": "week-01",
            "title": "项目周报",
            "rows": [
                {"owner": "Yu", "task": "接口联调", "status": "closed"},
                {"owner": "Luo", "task": "页面调整", "status": "processing"},
            ],
        },
        {
            "id": "week-02",
            "title": "资料整理",
            "rows": [{"owner": "Wen", "task": "目录更新", "status": "open"}],
        },
    ]
    fields = ["owner", "task", "status"]
    progress_file = "progress.txt"
    
    # Resume export if there are failed reports
    result = resume_export(reports, fields, progress_file)
    print(json.dumps(result, ensure_ascii=False, indent=2))

if __name__ == "__main__":
    main()
```

    progress_file = "progress.txt"
    
    # Simulate a failure
    with open(progress_file, "w") as f:

    assert len(result) == 1, "Should only export the remaining report"
    assert result[0]["id"] == "week-02", "Should export the remaining report"
