### 资料
为了处理较多报告的分批导出方案，我们将增加单报告失败记录、进度记录和中断后继续执行功能。我们将使用 Python 标准库来实现这些功能，并确保去重和恢复依据明确。

### 待确认信息
1. **报告是否需要去重**：是否需要在导出时检查报告的唯一性。
2. **进度记录**：如何记录和保存进度信息。
3. **中断后继续执行**：如何恢复中断的导出过程。

### 建议
1. **去重**：使用一个字典来存储已经导出的报告的 ID，以确保不会重复导出。
2. **进度记录**：使用 JSON 文件来记录导出进度。
3. **中断后继续执行**：通过读取进度文件来恢复中断的导出过程。

### 实现

```python
import csv
import io
import json
import os
import time

def render_csv(rows, fields):
    output = io.StringIO(newline="")
    writer = csv.DictWriter(output, fieldnames=fields, extrasaction="ignore")
    writer.writeheader()
    writer.writerows(rows)
    return output.getvalue()

def export_report(report, fields, exported_reports):
    if report["id"] in exported_reports:
        return False
    try:
        csv_content = render_csv(report["rows"], fields)
        exported_reports.add(report["id"])
        return True, csv_content
    except Exception as e:
        print(f"Failed to export report {report['id']}: {e}")
        return False, None

def save_progress(exported_reports, progress_file):
    with open(progress_file, "w") as f:
        json.dump(list(exported_reports), f)

def load_progress(progress_file):
    if os.path.exists(progress_file):
        with open(progress_file, "r") as f:
            return set(json.load(f))
    return set()

def export_reports(reports, fields, progress_file):
    exported_reports = load_progress(progress_file)
    for report in reports:
        success, csv_content = export_report(report, fields, exported_reports)
        if success:
            report["csv"] = csv_content
            exported_reports.add(report["id"])
        time.sleep(1)  # Simulate a delay to avoid overwhelming the system
    save_progress(exported_reports, progress_file)
    return list(exported_reports)

def main():
    reports = [
        {
            "id": "week-01",
            "title": "项目周报",
            "rows": [
                {"owner": "Yu", "task": "接口联调", "status": "closed"},
                {"owner": "Luo", "task": "页面调整", "status": "processing"},
            ],
        },
        {
            "id": "week-02",
            "title": "资料整理",
            "rows": [{"owner": "Wen", "task": "目录更新", "status": "open"}],
        },
    ]
    fields = ["owner", "task", "status"]
    progress_file = "export_progress.json"
    result = export_reports(reports, fields, progress_file)
    print(json.dumps(result, ensure_ascii=False, indent=2))

if __name__ == "__main__":
    main()
```
