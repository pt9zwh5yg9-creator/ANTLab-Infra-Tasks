### 资料
为了处理较多报告的分批导出方案，我们将增加单报告失败记录、进度记录和中断后继续执行功能。我们将使用 Python 标准库来实现这些功能，并确保去重和恢复依据明确。

### 待确认信息

### 建议
1. **使用唯一标识符**：使用 `report_id` 作为报告的唯一标识符。
2. **进度记录**：使用一个文件来记录处理进度，包括已处理的报告 ID 和当前处理的报告 ID。
3. **失败记录**：记录失败的报告 ID，并在恢复时重新处理这些报告。

### 实现

```python
import csv
import io
import json
import os
import time
from collections import defaultdict


def render_csv(rows, fields):
    output = io.StringIO(newline="")
    writer = csv.DictWriter(output, fieldnames=fields, extrasaction="ignore")
    writer.writeheader()
    writer.writerows(rows)
    return output.getvalue()


def export_report(report, fields, output_dir):
    report_id = report["id"]
    report_file = os.path.join(output_dir, f"{report_id}.csv")
    try:
        with open(report_file, 'w', newline='') as csvfile:
            writer = csv.DictWriter(csvfile, fieldnames=fields, extrasaction="ignore")
            writer.writeheader()
            writer.writerows(report["rows"])
        return True, report_id
    except Exception as e:
        print(f"Failed to export report {report_id}: {e}")
        return False, report_id


def export_reports(reports, fields, output_dir, progress_file):
    processed = set()
    failed = defaultdict(list)
    current_report_id = None

    if os.path.exists(progress_file):
        with open(progress_file, 'r') as f:
            progress = json.load(f)
            processed = set(progress.get('processed', []))
            failed = defaultdict(list, progress.get('failed', {}))
            current_report_id = progress.get('current_report_id')

    for report in reports:
        report_id = report["id"]
        if report_id in processed:
            continue
        if report_id == current_report_id:
            continue

        success, report_id = export_report(report, fields, output_dir)
        if success:
            processed.add(report_id)
        else:
            failed[report_id].append(report)

        with open(progress_file, 'w') as f:
            json.dump({
                'processed': list(processed),
                'failed': dict(failed),
                'current_report_id': current_report_id
            }, f, ensure_ascii=False, indent=2)

        if not success:
            current_report_id = report_id
            time.sleep(5)  # 小延迟，避免频繁请求

    return processed, failed


def main():
    reports = [
        {
            "id": "week-01",
            "title": "项目周报",
            "rows": [
                {"owner": "Yu", "task": "接口联调", "status": "closed"},
                {"owner": "Luo", "task": "页面调整", "status": "processing"},
            ],
        },
        {
            "id": "week-02",
            "title": "资料整理",
            "rows": [{"owner": "Wen", "task": "目录更新", "status": "open"}],
        },
    ]
    fields = ["owner", "task", "status"]
    output_dir = "output"
    progress_file = "progress.json"

    os.makedirs(output_dir, exist_ok=True)

    processed, failed = export_reports(reports, fields, output_dir, progress_file)
    print(f"Processed: {processed}")
    print(f"Failed: {failed}")


if __name__ == "__main__":
    main()
```

```python
def test_export_reports():
    reports = [
        {
            "id": "week-01",
            "title": "项目周报",
            "rows": [
                {"owner": "Yu", "task": "接口联调", "status": "closed"},
                {"owner": "Luo", "task": "页面调整", "status": "processing"},
            ],
        },
        {
            "id": "week-02",
            "title": "资料整理",
            "rows": [{"owner": "Wen", "task": "目录更新", "status": "open"}],
        },
    ]
    fields = ["owner", "task", "status"]
    output_dir = "test_output"
    progress_file = "test_progress.json"

    os.makedirs(output_dir, exist_ok=True)

    processed, failed = export_reports(reports, fields, output_dir, progress_file)
    assert len(processed) == 2
    assert len(failed) == 0

    # 模拟失败情况
    def mock_export_report(report, fields, output_dir):
        if report["id"] == "week-01":
            raise Exception("Failed to export report week-01")
        return True, report["id"]

    processed, failed = export_reports(reports, fields, output_dir, progress_file, mock_export_report)
    assert len(processed) == 1
    assert len(failed) == 1
    assert "week-01" in failed

    # 恢复情况
    processed, failed = export_reports(reports, fields, output_dir, progress_file, mock_export_report)
    assert len(processed) == 2
    assert len(failed) == 0

test_export_reports()
```

### 解释
1. **去重和恢复依据**：通过 `progress_file` 记录已处理的报告 ID 和当前处理的报告 ID，确保不会重复处理。
2. **进度记录**：在 `progress_file` 中记录处理进度，包括已处理的报告 ID 和当前处理的报告 ID。
3. **失败记录**：记录失败的报告 ID，并在恢复时重新处理这些报告。
