def aggregate_by_owner(records):
    """
    统计记录按处理人分类的总数，以及 open、processing、closed 三种状态的数量。
    :param records: 字典列表，每条记录包含 'owner' 和 'status'
    :return: 字典，包含统计结果
    """
    if not records:
        return {}
    
    result = {
        'unassigned': {'total': 0, 'open': 0, 'processing': 0, 'closed': 0, 'other': 0},
        'assigned': {'total': 0, 'open': 0, 'processing': 0, 'closed': 0, 'other': 0}
    }
    
    for record in records:
        owner = record.get('owner', 'unassigned').lower()
        status = record.get('status', '').lower()
        
        if owner == 'unassigned':
            result['unassigned']['total'] += 1
            if status == 'open':
                result['unassigned']['open'] += 1
            elif status == 'processing':
                result['unassigned']['processing'] += 1
            elif status == 'closed':
                result['unassigned']['closed'] += 1
            else:
                result['unassigned']['other'] += 1
        else:
            result['assigned']['total'] += 1
            if status == 'open':
                result['assigned']['open'] += 1
            elif status == 'processing':
                result['assigned']['processing'] += 1
            elif status == 'closed':
                result['assigned']['closed'] += 1
            else:
                result['assigned']['other'] += 1
    
    return result
