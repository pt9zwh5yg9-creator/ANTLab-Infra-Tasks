```python
def aggregate_by_owner(records):
    """
    统计记录按处理人分类的总数，以及 open、processing、closed 三种状态的数量。
    :param records: 字典列表，每条记录包含 'owner' 和 'status'。
    :return: 字典，包含 'total'、'open'、'processing'、'closed' 和 'other' 的统计结果。
    """
    if not records:
        return {}

    result = {
        'total': 0,
        'open': 0,
        'processing': 0,
        'closed': 0,
        'other': 0
    }

    for record in records:
        owner = record.get('owner', 'unassigned')
        status = record.get('status', '').lower()

        result['total'] += 1
        if status == 'open':
            result['open'] += 1
        elif status == 'processing':
            result['processing'] += 1
        elif status == 'closed':
            result['closed'] += 1
        else:
            result['other'] += 1

    return result

# 测试示例
test_records = [
    {'owner': 'Alice', 'status': 'Open'},
    {'owner': 'Bob', 'status': 'Processing'},
    {'owner': 'Charlie', 'status': 'Closed'},
    {'owner': '', 'status': 'open'},
    {'owner': 'Diana', 'status': 'unknown'},
    {'owner': 'Eve', 'status': 'closed'},
    {'owner': 'Fiona', 'status': 'processing'},
    {'owner': 'Gina', 'status': 'Open'},
    {'owner': 'Hank', 'status': 'closed'},
    {'owner': 'Ivy', 'status': 'unknown'},
    {'owner': 'Judy', 'status': 'processing'}
]

print(aggregate_by_owner(test_records))
```

### 测试示例
1. 空列表：
   ```python
   print(aggregate_by_owner([]))  # 输出: {}
   ```

2. 所有状态都存在：
   ```python
   print(aggregate_by_owner(test_records))  # 输出: {'total': 11, 'open': 3, 'processing': 4, 'closed': 3, 'other': 1}
   ```

3. 所有记录缺失或为空的 `owner`：
   ```python
   print(aggregate_by_owner([{'owner': '', 'status': 'open'}, {'owner': '', 'status': 'closed'}]))  # 输出: {'total': 2, 'open': 1, 'processing': 0, 'closed': 1, 'other': 0}
   ```

4. 所有记录状态为空：


5. 所有记录状态为 `unknown`：
   ```python
   print(aggregate_by_owner([{'owner': 'Alice', 'status': 'unknown'}, {'owner': 'Bob', 'status': 'unknown'}]))  # 输出: {'total': 2, 'open': 0, 'processing': 0, 'closed': 0, 'other': 2}
   ```
